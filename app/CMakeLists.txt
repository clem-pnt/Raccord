set(APP_SOURCES main.cpp)
if(WIN32)
  set(RACCORD_ICO ${CMAKE_SOURCE_DIR}/packaging/windows/raccord.ico)
  configure_file(${CMAKE_SOURCE_DIR}/packaging/windows/raccord.rc.in ${CMAKE_CURRENT_BINARY_DIR}/raccord.rc @ONLY)
  list(APPEND APP_SOURCES ${CMAKE_CURRENT_BINARY_DIR}/raccord.rc)
endif()
add_executable(raccord ${APP_SOURCES})
target_link_libraries(raccord PRIVATE raccord_ui)
if(WIN32)
  # GUI subsystem: no console window.
  set_target_properties(raccord PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
install(TARGETS raccord RUNTIME DESTINATION bin)
